### 生成算法实现

**基础生成逻辑**
```javascript
class USPhoneGenerator {
    constructor() {
        this.segments = this.loadSegments();
    }

    generate(options = {}) {
        const {
            count = 1,
            state = null,
            carrier = null,
            type = 'Any',
            exclude = []
        } = options;

        const numbers = [];
        for (let i = 0; i < count; i++) {
            const segment = this.selectRandomSegment(state, carrier, type);
            const number = this.formatNumber(segment);

            if (!exclude.includes(number)) {
                numbers.push(number);
            }
        }

        return numbers;
    }

    selectRandomSegment(state, carrier, type) {
        let segments = this.segments.filter(s => s.is_active);

        if (state) segments = segments.filter(s => s.state === state);
        if (carrier) segments = segments.filter(s => s.carrier === carrier);
        if (type !== 'Any') segments = segments.filter(s => s.type === type);

        // 避免选择特殊号码
        segments = segments.filter(s => {
            const npa = s.npa;
            const nxx = s.nxx;
            return !(npa === '555' || nxx === '555' ||
                    npa === '411' || nxx === '411');
        });

        return segments[Math.floor(Math.random() * segments.length)];
    }

    formatNumber(segment) {
        const npa = segment.npa;
        const nxx = segment.nxx;
        const line = this.generateLineNumber();
        return `+1${npa}${nxx}${line}`;
    }

    generateLineNumber() {
        // 生成用户号（XXXX）
        const line = Math.floor(Math.random() * 10000)
            .toString()
            .padStart(4, '0');
        return line;
    }
}
```

1. **同意原则**：必须获得接收方明确同意
2. **时间限制**：不得在上午8点前或晚上9点后拨打
3. **身份识别**：必须明确告知主叫身份
4. **退订机制**：必须提供便捷的退订方式
5. **记录保留**：保留通话同意记录至少4年

**违规处罚**

```javascript
function validateUSPhoneNumber(number) {
    // 移除所有非数字字符
    const digits = number.replace(/\D/g, '');

    // 检查长度
    if (digits.length !== 11 && digits.length !== 10) {
        return false;
    }

    // 检查国家代码
    if (digits.length === 11 && digits[0] !== '1') {
        return false;
    }

    // 提取NPA和NXX
    const startIndex = digits.length - 10;
    const npa = digits.slice(startIndex, startIndex + 3);
    const nxx = digits.slice(startIndex + 3, startIndex + 6);

    // 验证NPA和NXX规则
    if (npa[0] === '0' || npa[0] === '1' ||
        nxx[0] === '0' || nxx[0] === '1') {

    // 检查特殊代码
```
