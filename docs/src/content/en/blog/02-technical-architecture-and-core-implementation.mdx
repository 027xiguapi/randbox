### Core Module Details

#### 1. **Core Module - Infrastructure**

The Core module is the foundation of RandBox, providing basic random number generation capabilities and utility functions:

```typescript
// src/core.ts Core class design
export class RandBox {
  private mt: MersenneTwister;
  private seed: string | number | undefined;

  constructor(seed?: string | number) {
    this.seed = seed;
    this.mt = new MersenneTwister(this.getSeedValue(seed));
  }

  // Core random number generation method
  random(): number {
    return this.mt.random();
  }

  // Utility method: range check
  testRange(test: boolean, message: string): void {
    if (test) {
      throw new RangeError(message);
    }
  }

  // Utility method: initialize options
  initOptions<T>(options: T | undefined, defaultOptions: T): T {
    return { ...defaultOptions, ...options };
  }
}
```

**Why Choose Mersenne Twister?**

1. **High-Quality Randomness**: Period up to 2^19937-1, almost no repetition
2. **Excellent Performance**: High computational efficiency, suitable for large-scale data generation
3. **Predictability**: Supports seed configuration for easy testing and debugging
4. **Widely Recognized**: Adopted by many programming languages and libraries

```typescript
import { RandBox } from './core';

// Name generation function
export function name(this: RandBox, options?: PersonOptions): string {
  const opts = this.initOptions(options, {
    nationality: 'zh',
    gender: 'mixed',
    middle_initial: false
  });

  const firstNames = DATA_SETS.firstNames[opts.nationality][opts.gender];
  const lastNames = DATA_SETS.lastNames[opts.nationality];

  const firstName = this.pickone(firstNames);
  const lastName = this.pickone(lastNames);

  return opts.nationality === 'zh'
    ? `${lastName}${firstName}`
    : `${firstName} ${lastName}`;
}

// Email generation function
export function email(this: RandBox, options?: EmailOptions): string {
  const opts = this.initOptions(options, {
    domain: null,
    length: null
  });

  const username = opts.length
    ? this.string({ length: opts.length, casing: 'lower' })
    : this.first().toLowerCase();

  const domain = opts.domain || this.pickone(COMMON_DOMAINS);

  return `${username}@${domain}`;
}
```

#### 2. **Intelligent Caching Mechanism**

To improve performance, RandBox implements intelligent caching:

```typescript
class DataCache {
  private cache = new Map<string, any>();
  private maxSize = 1000;

  get<T>(key: string, generator: () => T): T {
    if (this.cache.has(key)) {
      return this.cache.get(key);
    }

    const value = generator();

    if (this.cache.size >= this.maxSize) {
      // LRU eviction strategy
      const firstKey = this.cache.keys().next().value;
      this.cache.delete(firstKey);
    }

    this.cache.set(key, value);
    return value;
  }
}
```

## Algorithm Core Implementation

### 1. **Mersenne Twister Implementation**

```typescript
class MersenneTwister {
  private mt: number[] = new Array(624);
  private mti: number = 625;

  constructor(seed?: number) {
    this.init_genrand(seed || Date.now());
  }

  private init_genrand(s: number): void {
    this.mt[0] = s >>> 0;
    for (this.mti = 1; this.mti < 624; this.mti++) {
      s = this.mt[this.mti - 1] ^ (this.mt[this.mti - 1] >>> 30);
      this.mt[this.mti] = (((((s & 0xffff0000) >>> 16) * 1812433253) << 16) +
                          (s & 0x0000ffff) * 1812433253) + this.mti;
      this.mt[this.mti] >>>= 0;
    }
  }

  genrand_int32(): number {
    let y: number;
    const mag01 = [0x0, 0x9908b0df];

    if (this.mti >= 624) {
      // Generate new random number sequence
      let kk: number;

      for (kk = 0; kk < 227; kk++) {
        y = (this.mt[kk] & 0x80000000) | (this.mt[kk + 1] & 0x7fffffff);
        this.mt[kk] = this.mt[kk + 397] ^ (y >>> 1) ^ mag01[y & 0x1];
      }

      // ... More implementation details
      this.mti = 0;
    }

    y = this.mt[this.mti++];
    // Tempering
    y ^= y >>> 11;
    y ^= (y << 7) & 0x9d2c5680;
    y ^= (y << 15) & 0xefc60000;
    y ^= y >>> 18;

    return y >>> 0;
  }

  random(): number {
    return this.genrand_int32() * (1.0 / 4294967296.0);
  }
}
```

### 2. **Advanced Random Algorithms**

#### Normal Distribution Implementation (Box-Muller Transform)

```typescript
export function normal(this: RandBox, mean = 0, deviation = 1): number {
  // Box-Muller transform for normal distribution
  if (this.normal_cache !== null) {
    const cached = this.normal_cache;
    this.normal_cache = null;
    return cached * deviation + mean;
  }

  const u1 = this.random();
  const u2 = this.random();

  const z0 = Math.sqrt(-2 * Math.log(u1)) * Math.cos(2 * Math.PI * u2);
  const z1 = Math.sqrt(-2 * Math.log(u1)) * Math.sin(2 * Math.PI * u2);

  this.normal_cache = z1;
  return z0 * deviation + mean;
}
```

#### Weighted Random Selection Implementation

```typescript
export function weighted<T>(
  this: RandBox,
  array: T[],
  weights: number[]
): T {
  if (array.length !== weights.length) {
    throw new Error('Array and weights must have the same length');
  }

  // Calculate cumulative weights
  const totalWeight = weights.reduce((sum, weight) => sum + weight, 0);
  const cumulativeWeights: number[] = [];

  let sum = 0;
  for (let i = 0; i < weights.length; i++) {
    sum += weights[i] / totalWeight;
    cumulativeWeights[i] = sum;
  }

  // Random selection
  const random = this.random();
  for (let i = 0; i < cumulativeWeights.length; i++) {
    if (random <= cumulativeWeights[i]) {
      return array[i];
    }
  }

  return array[array.length - 1];
}
```

### 1. **Lazy Loading Data Sets**

```typescript
class LazyDataSet<T> {
  private _data: T | null = null;
  private loader: () => T;

  constructor(loader: () => T) {
    this.loader = loader;
  }

  get data(): T {
    if (this._data === null) {
      this._data = this.loader();
    }
    return this._data;
  }
}
```

### 2. **Memory Pool Optimization**

```typescript
class ObjectPool<T> {
  private pool: T[] = [];
  private createFn: () => T;
  private resetFn: (obj: T) => void;

  constructor(createFn: () => T, resetFn: (obj: T) => void) {
    this.createFn = createFn;
    this.resetFn = resetFn;
  }

  get(): T {
    if (this.pool.length > 0) {
      return this.pool.pop()!;
    }
    return this.createFn();
  }

  release(obj: T): void {
    this.resetFn(obj);
    this.pool.push(obj);
  }
}
```
